1. Обзор системы

Функция handle_aux() — это главный обработчик вспомогательных LED (красный/зелёный). Она вызывается каждые 15мс из главного цикла main().

// main.c — цикл обработки
while (1) {
    if (mode == MODE_BEACON) {
        handle_aux();    // ← обработка каждые 15мс
        scan_ui1();
        Delay_Ms(15);   // ← интервал вызова
    }
}
Ключевое: Длительности duration0 и duration1 измеряются в количестве повторов по ~15мс.
Например: duration0 = 10 → 10 × 15мс = 150мс

3. Пошаговый алгоритм запуска

1 Инициализация переменных 
// Глобальные переменные (уже объявлены в aux_leds.h)
extern union AUX_Cmd_u aux_cmd_red;   // команда для красного
extern union AUX_Cmd_u aux_cmd_grn;   // команда для зелёного
extern AUX_Ctr_t aux_ctl_red;         // состояние красного
extern AUX_Ctr_t aux_ctl_grn;         // состояние зелёного

extern union AUX_Cmd_u *p_grn;        // текущий указатель зелёного
2 Сброс состояния (перед запуском новой программы) 
// Обнуляем контрольные структуры
uint32_t *p = (uint32_t*)&aux_ctl_red;
*p = 0;
p = (uint32_t*)&aux_ctl_grn;
*p = 0;

// Включаем обработку массивов

3 Способ A: Однократная команда (без массива) 
// Формируем 32-битную команду
uint32_t cmd = 0;

// Поля команды:
cmd |= RED_LED;            // биты 0..2: RED=1, GRN=2, HBL=4
cmd |= (0 << 3);           // бит 3: cycle (0=зациклено, 1=однократно)
cmd |= (5 << 4);           // биты 4..7: count (количество повторов)
cmd |= (10 << 8);          // биты 8..16: duration0 (в ×15мс)
cmd |= (5 << 17);          // биты 17..25: duration1 (в ×15мс)
cmd |= (1 << 26);          // биты 26..28: brightness0 (0=выкл, 1=ярко)
cmd |= (0 << 29);          // биты 29..31: brightness1

// Или короче через макрос:
// #define MAKE_CMD(dev, cyc, cnt, d0, d1, b0, b1) \
//     ((dev) | ((cyc)<<3) | ((cnt)<<4) | ((d0)<<8) | ((d1)<<17) | ((b0)<<26) | ((b1)<<29))

// Запускаем
set_aux_cmd(cmd, AUX_1S, true, IDM_OTHER);
4 Способ B: Программа из массива команд 
// Определяем массив команд
const uint32_t my_program[] = {
    // Команда 1: Красный, 10×15мс ЯРКО → 5×15мс ВЫКЛ, 3 повтора
    MAKE_CMD(RED_LED, 0, 3, 10, 5, 1, 0),

    // Команда 2: Зелёный, 5×15мс ЯРКО → 5×15мс ВЫКЛ, 2 повтора
    MAKE_CMD(GRN_LED, 0, 2, 5, 5, 1, 0),

    // Команда 3: Оба, бесконечно (cycle=0, count=0)
    MAKE_CMD(RED_LED | GRN_LED, 0, 0, 3, 3, 1, 0),

    // Конец: системная команда AC_STOP (все 32 бита == 0)
    0
};

// Устанавливаем указатель на начало
p_red = (union AUX_Cmd_u*)my_program;
p_grn = (union AUX_Cmd_u*)my_program;

// Сбрасываем индексы
aux_ctl_red.arr_index = 0;
aux_ctl_grn.arr_index = 0;
aux_ctl_red.period = PER_INIT;
aux_ctl_grn.period = PER_INIT;
aux_ctl_red.enable = true;
aux_ctl_grn.enable = true;

// Включаем обработку массивов
EnableArrayProcessing = true;
5 Вызов handle_aux() в цикле 
// handle_aux() должна вызываться каждые ~15мс
while (1) {
    handle_aux();
    Delay_Ms(15);
}

5. Продвинутое использование

5.1 Системные команды массива

КомандаЗначение (32 бита)Описание
AC_STOP0x00000000Остановить обработку массива
AC_REP_ATSTARTсм. нижеПовторить с начала массива
AC_REP_ATSTEPSсм. нижеПовторить с шагом

Формат системной команды:

// Биты 0..1: command (0=STOP, 1=REP_ATSTART, 2=REP_ATSTEPS)
// Биты 2..4: steps (отступ от начала для REP_ATSTEPS)
// Биты 5..31: ДОЛЖНЫ БЫТЬ = 0 (иначе это команда LED)

// Примеры:
#define CMD_STOP        0x00000000  // Остановить
#define CMD_REP_START   0x00000001  // Повторить с начала
#define CMD_REP_STEP2   0x00000005  // Повторить с шагом 2 (0b101)

5.2 Цепочка команд (SOS-паттерн)

// SOS на красном: . . . — — — . . .
// . = короткая (5×15мс), — = длинная (15×15мс)

    // Тире 2

    // Тире 3
    MAKE_CMD(RED_LED, 0, 1, 15, 15, 1, 0),
    // Точка 1

    // Точка 2

    // Точка 3 + пауза

6. Недостающие функции

Ниже — реализации функций, которые отсутствуют или закомментированы в проекте.

6.1 set_aux_cmd()

/**
 * @brief Установка однократной команды для красного LED
 * @param command  — 32-битная команда (см. union AUX_Cmd_u)
 * @param delay    — интервал таймера (AUX_15MS, AUX_60MS, AUX_1S и т.д.)
 * @param enable   — true = начать обработку, false = отложить
 * @param idm_mode — режим индикации (IDM_OTHER, IDM_BEACON и т.д.)
 */
void set_aux_cmd(uint32_t command, awu_tick_t delay, bool enable, IDM_Type_t idm_mode) {
    // Приводим 32-битный код к union AUX_Cmd_u
    aux_cmd_red.aux_cmd = command;

    // Устанавливаем контрольные флаги
    aux_ctl_red.enable = enable;
    aux_ctl_red.idmode = idm_mode;
    aux_ctl_red.period = PER_INIT;       // сброс периода на инициализацию
    aux_ctl_red.current_brightness = 0;  // сброс текущей яркости
    aux_ctl_red.work_duration = 0;       // сброс оставшейся длительности
    aux_ctl_red.work_count = 0;          // сброс счётчика повторов

    // Устанавливаем указатель на команду
    p_red = &aux_cmd_red;

    // Запускаем таймер (если нужно)
    if (delay != AUX_15MS) {
        awu_set_delay(delay);
    }
}

6.2 off_aux_cmd()

/**
 * @brief Остановка обработки AUX команд
 * @param stop_wdt — true = остановить таймер, false = оставить работать
 */
void off_aux_cmd(bool stop_wdt) {
    // Отключаем обработку
    aux_ctl_red.enable = false;
    aux_ctl_red.period = PER_INIT;
    aux_ctl_red.idmode = IDM_OTHER;

    // Обнуляем команду
    aux_cmd_red.aux_cmd = 0;
    p_red = &aux_cmd_red;

    // Выключаем LED
    smart_off_red();
    smart_off_green();

    // Останавливаем таймер если нужно
    if (stop_wdt) {
        awu_set_delay(AUX_15MS);  // минимальный интервал
    }
}

6.7 Макрос MAKE_CMD()

/**
 * @brief Макрос для формирования 32-битной команды
 * 
 * @param dev   — тип LED: RED_LED(1), GRN_LED(2), HBL_LED(4), RED|GRN(3)
 * @param cyc   — зацикленность: 0=бесконечно, 1=однократно
 * @param cnt   — количество повторов: 0..15 (0=бесконечно если cyc=0)
 * @param d0    — длительность 1-го полупериода: 0..511 (в ×15мс)
 * @param d1    — длительность 2-го полупериода: 0..511 (в ×15мс)
 * @param b0    — яркость 1-го полупериода: 0=выкл, 1=ярко, 2=слабо
 * @param b1    — яркость 2-го полупериода: 0=выкл, 1=ярко, 2=слабо
 * 
 * @return 32-битная команда для set_aux_cmd() или массива
 */
#define MAKE_CMD(dev, cyc, cnt, d0, d1, b0, b1) \
    ((uint32_t)((dev) & 0x7)          | \
     ((uint32_t)((cyc) & 0x1) << 3)   | \
     ((uint32_t)((cnt) & 0xF) << 4)   | \
     ((uint32_t)((d0) & 0x1FF) << 8)  | \
     ((uint32_t)((d1) & 0x1FF) << 17) | \
     ((uint32_t)((b0) & 0x7) << 26)   | \
     ((uint32_t)((b1) & 0x7) << 29))
Важно: Макрос MAKE_CMD() нужно добавить в aux_leds.h перед #endif.

7. Примеры из жизни

7.1 Информационное моргание (1 вспышка)

// Красный: 1 повтор, 1×15мс ЯРКО → 0×15мс ВЫКЛ
// Итого: 15мс вспышка
set_aux_cmd(INFO_RED, AUX_60MS, true, IDM_OTHER);

// Аналог через MAKE_CMD:
uint32_t cmd = MAKE_CMD(RED_LED, 1, 1, 1, 0, 2, 0);
set_aux_cmd(cmd, AUX_60MS, true, IDM_OTHER);

7.2 Beacon (бесконечное моргание)

// Beacon: красный, бесконечно, 8×15мс ЯРКО → 0×15мс ВЫКЛ
// Итого: 120мс вспышка каждые 120мс
void start_aux_beacon(void) {
    uint8_t beacon_type = ee.aux.beacon_type;
    uint16_t beacon_mode = ee.aux.beacon_mode;

    uint32_t cmd = 0b00000000000000010000000100001000
                   | beacon_type
                   | ((beacon_mode + 7) << 12);

    set_aux_cmd(cmd, AUX_1S, true, IDM_BEACON);
}

7.3 Предупреждение о низком напряжении

// Быстрое моргание: 2×15мс ЯРКО → 7×15мс ВЫКЛ
void start_exaux_voltage(WARNING_Fr_t ex_mode) {
    uint16_t low16 = 0;
    switch (ex_mode) {
        case VWARNING_FAST:    low16 = 0b0111000000001001; break;  // ~2с
        case VWARNING_MEDIUM:  low16 = 0b1000000000001011; break;  // ~4с
        case VWARNING_SLOW:    low16 = 0b1001000000001010; break;  // ~8с
    }
    set_aux_cmd(0b00000000000000010000000000000000 | low16, AUX_1S, true, IDM_VWARNING);
}

7.4 Счётчик импульсов (индикация заряда)

// Зелёный: count импульсов по 2×15мс
void aux_count(uint8_t count) {
    uint32_t cmd;
    if (count)
        cmd = 0b00000000000000010100000100000010 | (count << 4);
    else
        cmd = 0b00000000000000010010001000010001;  // одна красная если 0

    set_aux_cmd(cmd, AUX_250MS, true, IDM_OTHER);
}

8.3 Как расшифровать 32-битную команду

// Функция для отладки: вывод полей команды
void debug_cmd(uint32_t cmd) {
    union AUX_Cmd_u u;
    u.aux_cmd = cmd;

    EE_PRINTF("CMD: 0x%08lX\n", cmd);
    EE_PRINTF("  device    = %d\n", u.aux.device);
    EE_PRINTF("  cycle     = %d\n", u.aux.cycle);
    EE_PRINTF("  count     = %d\n", u.aux.count);
    EE_PRINTF("  duration0 = %d (%d ms)\n", u.aux.duration0, u.aux.duration0 * 15);
    EE_PRINTF("  duration1 = %d (%d ms)\n", u.aux.duration1, u.aux.duration1 * 15);
    EE_PRINTF("  bright0   = %d\n", u.aux.brightness0);
    EE_PRINTF("  bright1   = %d\n", u.aux.brightness1);
}
